%----------------------------------------
%---------- Mission sequence
%----------------------------------------

BeginMissionSequence;
minimumReboostFuel = fuelReserve + reboostFuelMargin;

% Do not issue a new "propagate to minAltitude" command after a reboost
% sequence has left the spacecraft below that altitude.  In that case the
% decaying orbit cannot cross minAltitude again and would propagate to entry.
% The margin prevents the two finite reboost impulses from consuming reserve fuel.
While DefaultSC.ChemicalTank1.FuelMass > minimumReboostFuel & DefaultSC.Earth.Altitude > minAltitude
   
   Propagate 'Propagate to Reboost Altitude' DefaultProp(DefaultSC) {DefaultSC.Earth.Altitude = minAltitude};
   
   Target 'Circular Reboost' DefaultDC {SolveMode = Solve, ExitMode = DiscardAndContinue, ShowProgressWindow = true};
      
      Propagate 'Propagate to Periapsis' DefaultProp(DefaultSC) {DefaultSC.Earth.Periapsis};
      
      % Burn magnitudes are in km/s.  Positive, bounded values prevent
      % retrograde solver trials from creating an Earth-intersecting orbit.
      Vary 'Vary TOI.V' DefaultDC(TOI.Element1 = 0.003, {Perturbation = 0.0001, Lower = 0, Upper = 0.02, MaxStep = 0.005, AdditiveScaleFactor = 0.0, MultiplicativeScaleFactor = 1.0});
      Vary 'Vary GOI.V' DefaultDC(GOI.Element1 = 0.003, {Perturbation = 0.0001, Lower = 0, Upper = 0.02, MaxStep = 0.005, AdditiveScaleFactor = 0.0, MultiplicativeScaleFactor = 1.0});
      
      Maneuver 'Apply TOI' TOI(DefaultSC);
      Propagate 'Propagate to Apoapsis' DefaultProp(DefaultSC) {DefaultSC.Earth.Apoapsis};
      Achieve 'Achieve Target SMA' DefaultDC(DefaultSC.Earth.SMA = targetSMA, {Tolerance = 1.0});
      
      Maneuver 'Apply GOI' GOI(DefaultSC);
      Achieve 'Achieve Circular Orbit' DefaultDC(DefaultSC.Earth.ECC = 0.0, {Tolerance = 0.005});
      
   EndTarget;  % For targeter DefaultDC
   
   Report ReboostReport DefaultSC.A1ModJulian DefaultSC.ChemicalTank1.FuelMass DefaultSC.Earth.Altitude;
   Report OrbitAnalysisReport DefaultSC.A1ModJulian DefaultSC.Earth.Altitude DefaultSC.ChemicalTank1.FuelMass DefaultSC.TotalMass DefaultSC.Earth.SMA DefaultSC.Earth.ECC DefaultSC.EarthMJ2000Eq.INC;

EndWhile;

% Controlled end-of-life descent.  This propagation terminates at the
% requested altitude (150 km by default), before the atmospheric model fails.
If DefaultSC.Earth.Altitude > finalAltitude
   Propagate 'Propagate to Final Altitude' DefaultProp(DefaultSC) {DefaultSC.Earth.Altitude = finalAltitude};
EndIf;

Report ReboostReport DefaultSC.A1ModJulian DefaultSC.ChemicalTank1.FuelMass DefaultSC.Earth.Altitude;
Report OrbitAnalysisReport DefaultSC.A1ModJulian DefaultSC.Earth.Altitude DefaultSC.ChemicalTank1.FuelMass DefaultSC.TotalMass DefaultSC.Earth.SMA DefaultSC.Earth.ECC DefaultSC.EarthMJ2000Eq.INC;
