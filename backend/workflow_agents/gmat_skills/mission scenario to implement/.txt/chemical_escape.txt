% MissionMode = 0 IADC-style graveyard transfer with Hohmann circularization.
% MissionMode = 1 direct Earth escape with a positive Earth-relative C3.

Create Variable MissionMode IADCBaseAltitude GraveyardAltitude GraveyardRadius EscapeC3;

BeginMissionSequence;

% --------------------------- USER PARAMETERS ---------------------------
% To use a GSO, edit SMAECCINCRAANAOPTA directly in Create Spacecraft.
% 0 = graveyard orbit, 1 = true Earth escape.
MissionMode = 0;
EscapeC3   = 0.01;
PostManeuverCoastDays = 2;
VisualizationCoastSeconds = 86400 * PostManeuverCoastDays;
% ----------------------------------------------------------------------

% IADC graveyard altitude (km) 235 + 1000  Cr  SRPArea  DryMass.
IADCBaseAltitude = 235;
GraveyardAltitude = IADCBaseAltitude + 1000 * GEO_CHEM.Cr * GEO_CHEM.SRPArea / GEO_CHEM.DryMass;
GEO_Radius = GEO_CHEM.Earth.SMA;
% For GEO or GSO, disposal is above the actual initial apogee plus IADC clearance.
GraveyardRadius = GEO_CHEM.Earth.RadApo + GraveyardAltitude;

InitialTotalMass = GEO_CHEM.TotalMass;
InitialFuelMass  = InitialTotalMass - GEO_CHEM.DryMass;

If 'Select chemical end-of-life mode' MissionMode < 0.5
   % First Target set transfer apogee to the desired disposal radius.
   Target 'Target chemical transfer apogee' GEO_CHEM_DC;
      Vary GEO_CHEM_DC(RaiseGraveyard.Element1 = 0.01, {Perturbation = 0.00001, Lower = 0, Upper = 0.1});
      Maneuver 'Inject Hohmann transfer to graveyard' RaiseGraveyard(GEO_CHEM);
      Propagate 'Coast to transfer apogee' GEO_CHEM_Prop(GEO_CHEM) {GEO_CHEM.Earth.Apoapsis};
      Achieve GEO_CHEM_DC(GEO_CHEM.Earth.RadApo = GraveyardRadius, {Tolerance = 0.01});
   EndTarget;
   % Second Target at that apogee, circularize to the same radius.
   Target 'Target chemical graveyard circularization' GEO_CHEM_DC;
      Vary GEO_CHEM_DC(CircularizeGraveyard.Element1 = 0.01, {Perturbation = 0.00001, Lower = 0, Upper = 0.1});
      Maneuver 'Circularize chemical graveyard orbit' CircularizeGraveyard(GEO_CHEM);
      Achieve GEO_CHEM_DC(GEO_CHEM.Earth.SMA = GraveyardRadius, {Tolerance = 0.01});
   EndTarget;
Else
   % Direct Earth escape with requested positive C3 in km^2s^2.
   Target 'Target direct chemical Earth escape' GEO_CHEM_DC;
      Vary GEO_CHEM_DC(EscapeBurn.Element1 = 1.3, {Perturbation = 0.0001, Lower = 0, Upper = 3});
      Maneuver 'Direct chemical Earth escape' EscapeBurn(GEO_CHEM);
      Achieve GEO_CHEM_DC(GEO_CHEM.Earth.C3Energy = EscapeC3, {Tolerance = 0.000001});
   EndTarget;
EndIf;

% Retain two final orbits in the plots, so the solved circular disposal orbit
% (or the initial escape arc) remains directly visible in GMAT.
Propagate 'Coast for visualization of final trajectory' GEO_CHEM_Prop(GEO_CHEM) {GEO_CHEM.ElapsedSecs = VisualizationCoastSeconds};

FinalTotalMass = GEO_CHEM.TotalMass;
FinalFuelMass  = FinalTotalMass - GEO_CHEM.DryMass;
PropellantUsed = InitialFuelMass - FinalFuelMass;
DeltaV = 0;
DeltaV = RaiseGraveyard.Element1 + CircularizeGraveyard.Element1 + EscapeBurn.Element1;
Report GEO_CHEM_Summary MissionMode GEO_CHEM.ElapsedDays InitialFuelMass FinalFuelMass PropellantUsed DeltaV GEO_CHEM.Earth.SMA GEO_CHEM.Earth.RadPer GEO_CHEM.Earth.RadApo GEO_CHEM.Earth.ECC GEO_CHEM.Earth.C3Energy;
