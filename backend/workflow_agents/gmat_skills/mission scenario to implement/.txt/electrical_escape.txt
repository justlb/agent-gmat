%----------------------------------------
%---------- Mission Sequence
%----------------------------------------


BeginMissionSequence;

% --------------------------- USER PARAMETERS ---------------------------
% 0 = graveyard orbit, 1 = true Earth escape.
MissionMode = 1;
% Stop before depletion; reaching this reserve means the requested disposal target was not achieved.
FuelReserveKg = 0.01;
% Positive Earth-relative C3 (km^2/s^2) means an unbound Earth trajectory.
EscapeC3 = 0.01;
MaxCemeteryDays = 20;
MaxEscapeDays = 300;
% ----------------------------------------------------------------------

% IADC disposal rule (km): 235 + 1000 * Cr * (A/m).
% A is the SRP reference area (m^2), m the dry mass (kg).  The resulting
% altitude is the required minimum perigee altitude above GEO.
IADCBaseAltitude = 235;
GraveyardAltitude = IADCBaseAltitude + 1000 * GEO_ESC.Cr * GEO_ESC.SRPArea / GEO_ESC.DryMass;
GraveyardRadius = 42164.17 + GraveyardAltitude;
% Avoid classifying a numerical stop-point roundoff as a disposal failure.
DisposalRadiusToleranceKm = 0.1;
DisposalC3Tolerance = 1e-5;
GraveyardRadiusAcceptance = GraveyardRadius - DisposalRadiusToleranceKm;
EscapeC3Acceptance = EscapeC3 - DisposalC3Tolerance;
Toggle EphemerisFile1 On;
DisposalTargetReached = 0;
DisposalLimitReached = 0;
Report GEO_ESC_Report GEO_ESC.UTCGregorian GEO_ESC.ElapsedDays GEO_ESC.Earth.SMA GEO_ESC.Earth.RadPer GEO_ESC.Earth.RadApo GEO_ESC.Earth.ECC GEO_ESC.Earth.C3Energy GEO_ESC.TotalMass DisposalTargetReached DisposalLimitReached;

If 'Select end-of-life mode' MissionMode < 0.5
   % Continuous prograde thrust raises the GEO orbit.  The first condition
   % reached ends the burn, so MaxCemeteryDays is a safety limit.
   BeginFiniteBurn 'Raise to graveyard orbit' ElectricSpiral(GEO_ESC);
   Propagate 'Electric spiral until perigee clears GEO' GEO_ESC_Prop(GEO_ESC) {GEO_ESC.Earth.RadPer = GraveyardRadius, GEO_ESC.ElapsedDays = MaxCemeteryDays, GEO_ESC.ElectricPropellantTank_ESC.FuelMass = FuelReserveKg};
   EndFiniteBurn 'Stop at graveyard altitude' ElectricSpiral(GEO_ESC);
   If 'Graveyard target reached' GEO_ESC.Earth.RadPer >= GraveyardRadiusAcceptance
      DisposalTargetReached = 1;
   Else
      DisposalLimitReached = 1;
   EndIf;
Else
   % C3 > 0 is the physical Earth-escape condition.  This remains an
   % Earth-centred model; a planetary destination would need a new design.
   BeginFiniteBurn 'Spiral to Earth escape energy' ElectricSpiral(GEO_ESC);
   Propagate 'Electric spiral until positive C3' GEO_ESC_Prop(GEO_ESC) {GEO_ESC.Earth.C3Energy = EscapeC3, GEO_ESC.ElapsedDays = MaxEscapeDays, GEO_ESC.ElectricPropellantTank_ESC.FuelMass = FuelReserveKg};
   EndFiniteBurn 'Stop after Earth escape criterion' ElectricSpiral(GEO_ESC);
   If 'Escape target reached' GEO_ESC.Earth.C3Energy >= EscapeC3Acceptance
      DisposalTargetReached = 1;
   Else
      DisposalLimitReached = 1;
   EndIf;
EndIf;

Report GEO_ESC_Report GEO_ESC.UTCGregorian GEO_ESC.ElapsedDays GEO_ESC.Earth.SMA GEO_ESC.Earth.RadPer GEO_ESC.Earth.RadApo GEO_ESC.Earth.ECC GEO_ESC.Earth.C3Energy GEO_ESC.TotalMass DisposalTargetReached DisposalLimitReached;
Toggle EphemerisFile1 Off;
